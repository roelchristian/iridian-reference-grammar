\newcommand{\bookcover}[1][1]{%
  \clearpage
  \begingroup
  \normalfont\normalsize
  \thispagestyle{empty}%
  \null
  \begin{tikzpicture}[remember picture,overlay]
    \begin{scope}[shift={(current page.north west)},x=1mm,y=-1mm]
      \begin{scope}[shift={(\ircoveroffset,\ircoveroffset)}]
        \ifcase#1\relax
        \or\ircoverprimary
        \or\ircovermargin
        \or\ircoverstrata
        \else\ircoverprimary
        \fi
      \end{scope}
    \end{scope}
  \end{tikzpicture}%
  \par\clearpage
  \endgroup
}
